# api_up <repo-dir> [stub-delay-ms] : bootJar, start stub(9090, delayed) and api(8080), wait until api answers
api_up() {
  (cd "$1" && ./gradlew :apps:api:bootJar --console=plain -q 2>&1 | grep -E 'FAIL|error:' || true)
  (cd $ROOT/insurance-platform-poc && exec env STUB_DELAY_MS=${2:-0} python3 -I tools/stub-server.py 9090 >/dev/null 2>&1) & echo $! > $S/stub.pid

  for i in $(seq 1 40); do curl -s -o /dev/null localhost:8080/contracts/x && return 0; sleep 1; done
  echo "api did not start (see $S/api.log)"; return 1

api_down() { kill $(cat $S/api.pid) $(cat $S/stub.pid) 2>/dev/null; sleep 1; pkill -f 'stub-server.py 9090' 2>/dev/null; true; }

# quote_probe : POST a contract then GET its quote; prints "<http_code> <seconds>"
quote_probe() {
  local id
  id=$(curl -s -X POST localhost:8080/contracts -H 'Content-Type: application/json' \
    -d '{"holderName":"Kim","beneficiary":"Park","insuredAmount":10000000,"startDate":"2026-01-01","endDate":"2027-01-01"}' \
    | sed 's/.*"id":"\([^"]*\)".*/\1/')
  curl -s -o /dev/null -w '%{http_code} %{time_total}\n' "localhost:8080/contracts/$id/quote"
}

# pr_merge <repo> <branch> : push branch, open PR, wait for checks, merge, return to main. Prints PR number.
pr_merge() {
  local repo=$1 br=$2 pr
  git push -q -u origin "$br"
  pr=$(gh pr create -R khkim6040/$repo --fill --head "$br" | grep -o '[0-9]*$')
  gh pr checks "$pr" -R khkim6040/$repo --watch --fail-fast >/dev/null
  gh pr merge "$pr" -R khkim6040/$repo --merge --delete-branch >/dev/null
  git checkout -q main; git pull -q --ff-only origin main; git branch -D "$br" >/dev/null 2>&1 || true
  echo "$pr"
}

# run_wait <repo> <workflow> : wait for the newest run of a workflow, print its URL and conclusion
run_wait() {
  local id; sleep 12
  id=$(gh run list -R khkim6040/$1 --workflow "$2" --limit 1 --json databaseId --jq '.[0].databaseId')
  gh run watch "$id" -R khkim6040/$1 --exit-status >/dev/null && echo "ok  https://github.com/khkim6040/$1/actions/runs/$id" || echo "FAIL https://github.com/khkim6040/$1/actions/runs/$id"
}

# tests_total <repo-dir> : sum of junit test counts
tests_total() { grep -ho 'tests="[0-9]*"' "$1"/*/*/build/test-results/test/*.xml "$1"/*/build/test-results/test/*.xml 2>/dev/null | grep -o '[0-9]*' | paste -sd+ - | bc; }
EOF

package io.github.khkim6040.platform.http

object CorrelationId {
    const val HEADER = "X-Correlation-Id"
    const val SOURCE_HEADER = "X-Correlation-Source"
    const val MDC_KEY = "correlationId"
}
EOF
  cat > $P/servlet/CorrelationIdFilter.kt <<'EOF'
package io.github.khkim6040.platform.http.servlet

import io.github.khkim6040.platform.http.CorrelationId
import jakarta.servlet.FilterChain
import jakarta.servlet.http.HttpServletRequest
import jakarta.servlet.http.HttpServletResponse
import org.slf4j.MDC
import org.springframework.web.filter.OncePerRequestFilter
import java.util.UUID

class CorrelationIdFilter : OncePerRequestFilter() {
    override fun doFilterInternal(request: HttpServletRequest, response: HttpServletResponse, chain: FilterChain) {
        val inbound = request.getHeader(CorrelationId.HEADER)?.takeIf { it.isNotBlank() }
        val id = inbound ?: UUID.randomUUID().toString()
        MDC.put(CorrelationId.MDC_KEY, id)
        response.setHeader(CorrelationId.HEADER, id)
        response.setHeader(CorrelationId.SOURCE_HEADER, if (inbound == null) "generated" else "propagated")
        try {
            chain.doFilter(request, response)
        } finally {
            MDC.remove(CorrelationId.MDC_KEY)
        }
    }
}
EOF
